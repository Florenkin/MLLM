cmake_minimum_required(VERSION 3.25)
include(cmake/Environment.cmake)
project(MLLM VERSION 0.1.0 LANGUAGES CXX)
if(MSVC AND DEFINED MLLM_MSVC_INCLUDE_PREFIX)
  # CMake's localized include-prefix detection uses the system code page;
  # Ninja receives UTF-8 from this toolchain. Keep header dependencies intact.
  set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${MLLM_MSVC_INCLUDE_PREFIX}  ")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${CMAKE_CXX_CL_SHOWINCLUDES_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  add_compile_options(/utf-8 /EHsc /bigobj /W4 "/FI${CMAKE_SOURCE_DIR}/cmake/Qt5MsvcCompatibility.h")
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc imgcodecs calib3d videoio)
option(MLLM_ENABLE_HIK "Enable Hikvision MVS adapter" OFF)
add_library(mllm_core src/core/IO.cpp src/core/Laser.cpp src/core/Calibration.cpp src/core/Reconstruction.cpp src/core/PointCloud.cpp)
target_include_directories(mllm_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(mllm_core PUBLIC ${OpenCV_LIBS})
add_library(mllm_services src/app/services/Workflow.cpp src/app/acquisition/Camera.cpp)
target_link_libraries(mllm_services PUBLIC mllm_core Qt5::Core)
if(MLLM_ENABLE_HIK)
  find_path(MVS_INCLUDE MvCameraControl.h PATHS "${HIK_MVS_ROOT}/Development/Includes" REQUIRED NO_DEFAULT_PATH)
  find_library(MVS_LIBRARY MvCameraControl PATHS "${HIK_MVS_ROOT}/Development/Libraries/win64" REQUIRED NO_DEFAULT_PATH)
  target_include_directories(mllm_services PRIVATE "${MVS_INCLUDE}")
  target_link_libraries(mllm_services PRIVATE "${MVS_LIBRARY}")
  target_compile_definitions(mllm_services PRIVATE MLLM_WITH_HIK=1)
endif()
add_executable(MLLM WIN32 src/app/main.cpp src/app/ui/MainWindow.cpp src/app/ui/Views.cpp)
target_link_libraries(MLLM PRIVATE mllm_services Qt5::Widgets Qt5::Concurrent)
add_custom_command(TARGET MLLM POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -DAPP=$<TARGET_FILE:MLLM> -DQT_DIR=${Qt5_DIR} -DCV_DIR=${OpenCV_DIR} -DCONFIG=$<CONFIG> -DHIK_RUNTIME=${HIK_MVS_RUNTIME_DIR} -DHIK_ENABLED=${MLLM_ENABLE_HIK} -P "${CMAKE_SOURCE_DIR}/cmake/Deploy.cmake"
  COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:MLLM>/config"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_SOURCE_DIR}/config/defaults/settings.ini" "$<TARGET_FILE_DIR:MLLM>/config/defaults.ini"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_SOURCE_DIR}/config/qt.conf" "$<TARGET_FILE_DIR:MLLM>/qt.conf")
include(CTest)
if(BUILD_TESTING)
  add_executable(mllm_tests test/CoreTests.cpp)
  target_link_libraries(mllm_tests PRIVATE mllm_services)
  add_dependencies(mllm_tests MLLM)
  add_test(NAME geometry_and_workflow COMMAND mllm_tests)
  add_test(NAME ui_smoke COMMAND MLLM --smoke-test)
  add_executable(mllm_ui_tests test/UiTests.cpp src/app/ui/MainWindow.cpp src/app/ui/Views.cpp)
  target_link_libraries(mllm_ui_tests PRIVATE mllm_services Qt5::Widgets Qt5::Concurrent)
  add_dependencies(mllm_ui_tests mllm_tests)
  add_test(NAME ui_workflow COMMAND mllm_ui_tests)
  set_tests_properties(ui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=$ENV{SystemRoot}/Fonts" TIMEOUT 90)
  set_tests_properties(geometry_and_workflow PROPERTIES TIMEOUT 90)
  set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=$ENV{SystemRoot}/Fonts;MLLM_CONFIG_DIR=${CMAKE_BINARY_DIR}/test-config" TIMEOUT 30)
endif()
install(TARGETS MLLM RUNTIME DESTINATION .)
install(DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/" DESTINATION . FILES_MATCHING PATTERN "*.dll" PATTERN "*.exe" EXCLUDE)
install(FILES config/defaults/settings.ini DESTINATION config RENAME defaults.ini)
install(FILES README.md TASKS.md VALIDATION.md DESTINATION .)
install(FILES LaunchDemo.ps1 DESTINATION .)
install(FILES config/qt.conf DESTINATION .)
